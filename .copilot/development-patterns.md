### API Resource Pattern
```csharp
// ✅ Standard API resource pattern
public partial class TemplateApiResource : ApiResource
{
    // Private parameterless constructor for JSON deserialization
    private TemplateApiResource() { }
    
    // Public constructor with all required properties
    public TemplateApiResource(string id, string name, string content)
    {
        this.Id = id;
        this.Name = name;
        this.Content = content;
    }
    
    // Properties with private setters and null-forgiving operators
    public string Id { get; private set; } = null!;
    public string Name { get; private set; } = null!;
    public string Content { get; private set; } = null!;
    
    // Optional properties should be nullable
    public string? Description { get; private set; }
    public DateTime? LastModified { get; private set; }
}
```

## Attribute Usage Patterns

### HTTP Method Attributes

### Link Attributes

## Error Handling Patterns

### Operation Results
```csharp
// ✅ Use standard operation result types
public Task<OperationResult> HandleAsync(MyQuery query)
{
    try
    {
        var result = ProcessQuery(query);
        return Task.FromResult<OperationResult>(new OkResult(result));
    }
    catch (ValidationException ex)
    {
        return Task.FromResult<OperationResult>(new BadRequestResult(ex.Message));
    }
    catch (NotFoundException ex)
    {
        return Task.FromResult<OperationResult>(new NotFoundResult(ex.Message));
    }
}
```

## Integration with ASP.NET Core

### Startup Configuration
```csharp
// ✅ Standard startup pattern
public class Startup
{
    public void ConfigureServices(IServiceCollection services)
    {
        // Register Blueprint services
        services.AddSingleton<MyBlueprintAppExecutor>();
        services.AddTransient<IMyService, MyService>();
    }
    
    public void Configure(IApplicationBuilder app, IWebHostEnvironment env)
    {
        // Configure Blueprint endpoints
        app.UseRouting();
        app.UseAuthentication();
        app.UseAuthorization();
        
        app.UseEndpoints(endpoints =>
        {
            endpoints.MapBlueprintOperations<MyBlueprintAppExecutor>();
        });
    }
}
```
